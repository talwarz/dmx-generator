      let channels = [];
      let activePage = 1;
      let wheelAssignments = {};
      const categories = ['Intensity', 'Colour', 'C-Mix', 'Gobo 1', 'Gobo 2', 'Beam', 'Effect', 'Special', 'Control'];
      const functionOptions = ['Dimmer','Shutter','Red','Green','Blue','White','Amber','UV','Pan','Pan Fine','Tilt','Tilt Fine','Colour Func','Colour','Color Macro','Gobo Func','Gobo','Gobo Rotate','Prism 1','Prism 1 Rot','Focus','Frost','Zoom','P/T Speed','Control','Reset','Lamp On/Off','Auto Program','Sound Mode','Custom Function'];
      function blankPage() {
      const page = {};
      categories.forEach(cat => page[cat] = { w1: '', w2: '', w3: '' });
      return page;
      }
      function ensurePage(pageNo) {
      if (!wheelAssignments[pageNo]) wheelAssignments[pageNo] = blankPage();
      }
      function updateTotal() { document.getElementById('totalChannels').value = channels.length; }
      function renderChannels() {
      const body = document.getElementById('channelBody');
      body.innerHTML = '';
      channels.forEach((ch, index) => {
      const tr = document.createElement('tr');
      tr.innerHTML = `
      <td>${index + 1}</td>
      <td class="function-cell">
      <select onchange="setChannelFunction(${index}, this.value)">${buildFunctionOptions(ch.name)}</select>
      <input style="margin-top:8px;" placeholder="Custom function name" value="${escapeHtml(ch.name)}" oninput="setChannelFunction(${index}, this.value)">
      </td>
      <td class="small-col"><input type="number" value="${ch.default}" min="0" max="255" onchange="channels[${index}].default=this.value"></td>
      <td class="small-col"><input type="number" value="${ch.min}" min="0" max="255" onchange="channels[${index}].min=this.value"></td>
      <td class="small-col"><input type="number" value="${ch.max}" min="0" max="255" onchange="channels[${index}].max=this.value"></td>
      <td><button class="danger" onclick="removeChannel(${index})">Remove</button></td>`;
      body.appendChild(tr);
      });
      updateTotal();
      renderAssignment();
      }
      function buildFunctionOptions(selectedValue) {
      let html = '';
      functionOptions.forEach(fn => html += `<option value="${escapeHtml(fn)}" ${fn === selectedValue ? 'selected' : ''}>${escapeHtml(fn)}</option>`);
      if (selectedValue && !functionOptions.includes(selectedValue)) html += `<option value="${escapeHtml(selectedValue)}" selected>${escapeHtml(selectedValue)}</option>`;
      return html;
      }
      function buildAssignmentOptions(selectedValue) {
      const names = [...new Set(channels.map(ch => String(ch.name || '').trim()).filter(Boolean))];
      let html = `<option value=""> </option>`;
      names.forEach(name => html += `<option value="${escapeHtml(name)}" ${name === selectedValue ? 'selected' : ''}>${escapeHtml(name)}</option>`);
      if (selectedValue && !names.includes(selectedValue)) html += `<option value="${escapeHtml(selectedValue)}" selected>${escapeHtml(selectedValue)}</option>`;
      return html;
      }
      function setChannelFunction(index, value) {
      channels[index].name = value;
      renderAssignment();
      }
      function addChannel(name = 'Dimmer', def = 0, min = 0, max = 255) {
      channels.push({ name, default: def, min, max });
      renderChannels();
      }
      function removeChannel(index) {
      channels.splice(index, 1);
      renderChannels();
      }
      function clearChannels() {
      channels = [];
      wheelAssignments = {};
      activePage = 1;
      document.getElementById('currentPage').value = 1;
      renderChannels();
      document.getElementById('preview').textContent = 'Preview yahan dikhega...';
      }
      function loadSample() {
      channels = [
      { name: 'Dimmer', default: 0, min: 0, max: 255 },
      { name: 'Shutter', default: 0, min: 0, max: 255 },
      { name: 'Red', default: 0, min: 0, max: 255 },
      { name: 'Green', default: 0, min: 0, max: 255 },
      { name: 'Blue', default: 0, min: 0, max: 255 },
      { name: 'Colour Func', default: 0, min: 0, max: 255 },
      { name: 'Colour', default: 0, min: 0, max: 255 },
      { name: 'Focus', default: 0, min: 0, max: 255 },
      { name: 'Frost', default: 0, min: 0, max: 255 },
      { name: 'Prism 1', default: 0, min: 0, max: 255 },
      { name: 'Prism 1 Rot', default: 0, min: 0, max: 255 },
      { name: 'P/T Speed', default: 0, min: 0, max: 255 },
      { name: 'Control', default: 0, min: 0, max: 255 }
      ];
      activePage = 1;
      document.getElementById('currentPage').value = 1;
      wheelAssignments = {};
      ensurePage(1);
      wheelAssignments[1]['Intensity'] = { w1: 'Shutter', w2: '', w3: '' };
      wheelAssignments[1]['Colour'] = { w1: 'Colour Func', w2: 'Colour', w3: '' };
      wheelAssignments[1]['C-Mix'] = { w1: 'Red', w2: 'Green', w3: 'Blue' };
      wheelAssignments[1]['Beam'] = { w1: '', w2: 'Focus', w3: 'Frost' };
      wheelAssignments[1]['Effect'] = { w1: '', w2: 'Prism 1', w3: 'Prism 1 Rot' };
      wheelAssignments[1]['Special'] = { w1: 'P/T Speed', w2: '', w3: '' };
      wheelAssignments[1]['Control'] = { w1: 'Control', w2: '', w3: '' };
      renderChannels();
      }
      function renderAssignment() {
      ensurePage(activePage);
      const body = document.getElementById('assignmentBody');
      body.innerHTML = '';
      categories.forEach(cat => {
      const row = wheelAssignments[activePage][cat];
      const tr = document.createElement('tr');
      tr.innerHTML = `
      <td class="category-label">${cat}</td>
      <td><select onchange="setAssignment('${cat}', 'w1', this.value)">${buildAssignmentOptions(row.w1)}</select></td>
      <td><select onchange="setAssignment('${cat}', 'w2', this.value)">${buildAssignmentOptions(row.w2)}</select></td>
      <td><select onchange="setAssignment('${cat}', 'w3', this.value)">${buildAssignmentOptions(row.w3)}</select></td>`;
      body.appendChild(tr);
      });
      renderUnassigned();
      validateWheelMapping();
      }
      function setAssignment(category, wheelKey, value) {
      ensurePage(activePage);
      wheelAssignments[activePage][category][wheelKey] = value;
      renderUnassigned();
      validateWheelMapping();
      }
      function renderUnassigned() {
      const assigned = new Set();
      const page = wheelAssignments[activePage] || blankPage();
      Object.values(page).forEach(row => Object.values(row).forEach(v => { if (v) assigned.add(v); }));
      const box = document.getElementById('unassignedBox');
      const names = [...new Set(channels.map(ch => String(ch.name || '').trim()).filter(Boolean))];
      const unassigned = names.filter(name => !assigned.has(name));
      box.innerHTML = unassigned.length ? unassigned.map(name => `<div class="unassigned-item">${escapeHtml(name)}</div>`).join('') : '<div class="unassigned-item">All functions assigned on this page</div>';
      }
      function changePage(delta) { setPage(activePage + delta); }
      function setPage(value) {
      let page = parseInt(value, 10);
      if (!page || page < 1) page = 1;
      if (page > 20) page = 20;
      activePage = page;
      document.getElementById('currentPage').value = page;
      ensurePage(page);
      renderAssignment();
      }
      function validateWheelMapping() {
      const errorBox = document.getElementById('errorBox');
      errorBox.style.display = 'none';
      errorBox.innerHTML = '';
      return true;
      }
      function getFixtureData() {
      return {
      brand: document.getElementById('brand').value.trim() || 'Unknown Brand',
      fixtureName: document.getElementById('fixtureName').value.trim() || 'Custom Fixture',
      modeName: document.getElementById('modeName').value.trim() || 'Default Mode',
      notes: document.getElementById('notes').value.trim(),
      channels,
      wheelAssignments
      };
      }
      function makeXML() {
      const data = getFixtureData();
      let xml = `<?xml version="1.0" encoding="UTF-8"?>\n<fixture>\n`;
      xml += ` <manufacturer>${xmlEscape(data.brand)}</manufacturer>\n <name>${xmlEscape(data.fixtureName)}</name>\n <mode name="${xmlEscape(data.modeName)}" channels="${data.channels.length}">\n`;
      data.channels.forEach((ch, i) => {
      xml += ` <channel number="${i + 1}">\n <function>${xmlEscape(ch.name)}</function>\n <default>${ch.default}</default>\n <range min="${ch.min}" max="${ch.max}" />\n </channel>\n`;
      });
      xml += ` </mode>\n <wheelAssignment>\n`;
      Object.keys(data.wheelAssignments).sort((a,b)=>Number(a)-Number(b)).forEach(pageNo => {
      xml += ` <page number="${pageNo}">\n`;
      categories.forEach(cat => {
      const row = data.wheelAssignments[pageNo][cat] || { w1:'', w2:'', w3:'' };
      xml += ` <category name="${xmlEscape(cat)}" wheel1="${xmlEscape(row.w1)}" wheel2="${xmlEscape(row.w2)}" wheel3="${xmlEscape(row.w3)}" />\n`;
      });
      xml += ` </page>\n`;
      });
      xml += ` </wheelAssignment>\n <notes>${xmlEscape(data.notes)}</notes>\n</fixture>`;
      return xml;
      }
      function makeD4() {
      const data = getFixtureData();
      let text = `# D4 Fixture Personality Generated File\n# Manufacturer: ${data.brand}\n# Fixture: ${data.fixtureName}\n# Mode: ${data.modeName}\n# Channels: ${data.channels.length}\n\n[Fixture]\nManufacturer=${data.brand}\nName=${data.fixtureName}\nMode=${data.modeName}\nChannels=${data.channels.length}\n\n[Channels]\n`;
      data.channels.forEach((ch, i) => text += `${i + 1}=${ch.name},Default:${ch.default},Min:${ch.min},Max:${ch.max}\n`);
      Object.keys(data.wheelAssignments).sort((a,b)=>Number(a)-Number(b)).forEach(pageNo => {
      text += `\n[Wheel_Assignment_Page_${pageNo}]\n`;
      categories.forEach(cat => {
      const row = data.wheelAssignments[pageNo][cat] || { w1:'', w2:'', w3:'' };

      function makeR20() {
      const data = getFixtureData();
      let text = `R20 FIXTURE PROFILE\nBRAND:${data.brand}\nFIXTURE:${data.fixtureName}\nMODE:${data.modeName}\nCHANNELS:${data.channels.length}\n\n`;
      data.channels.forEach((ch, i) => text += `CH${i + 1}|${ch.name}|DEFAULT=${ch.default}|MIN=${ch.min}|MAX=${ch.max}\n`);
      Object.keys(data.wheelAssignments).sort((a,b)=>Number(a)-Number(b)).forEach(pageNo => {
      text += `\nWHEEL_ASSIGNMENT_PAGE_${pageNo}\n`;

      });
      });
      text += `\nNOTES:${data.notes}\n`;
